const esc = (t) => String(t == null ? '' : t).replace(/[&<>"]/g, (c) => ({'&':'&','<':'<','>':'>','"':'"'}[c])); const read = (k) => { try { return localStorage.getItem(k) === '1' } catch (e) { return false } }; const write = (k, v) => { try { v ? localStorage.setItem(k, '1') : localStorage.removeItem(k) } catch (e) {} }; const pkey = (p) => 'gdc-desk-posted-' + p.n; /* Copy as PLAIN TEXT, always. Justin, 6 November 2026: copying from this page was arriving with a black background. The cause was the old fallback, which selected the styled 
   and copied it as rich HTML, so in dark mode the clipboard carried
   background:#1D1A12 into LinkedIn and Substack.
   A textarea's value is plain text by definition, so execCommand('copy') from a
   hidden textarea can never carry styling. That is the fallback now. The last
   resort reveals a plain white textarea to copy from by hand, never the 
. */
async function copyPlain(text, btn) {
  if (!btn.dataset.label) btn.dataset.label = btn.textContent;
  const done = (msg) => {
    btn.textContent = msg;
    setTimeout(() => { btn.textContent = btn.dataset.label }, 1800);
  };
  try {
    if (navigator.clipboard && navigator.clipboard.writeText) {
      await navigator.clipboard.writeText(text);
      return done('Copied');
    }
  } catch (e) { /* fall through, this throws in app webviews and insecure contexts */ }
  try {
    const ta = document.createElement('textarea');
    ta.value = text;
    ta.setAttribute('readonly', '');
    ta.style.cssText = 'position:fixed;top:0;left:0;width:1px;height:1px;opacity:0;pointer-events:none';
    document.body.appendChild(ta);
    ta.focus();
    ta.select();
    ta.setSelectionRange(0, text.length);
    const ok = document.execCommand && document.execCommand('copy');
    document.body.removeChild(ta);
    if (ok) return done('Copied');
  } catch (e) { /* fall through */ }
  const host = btn.closest('.copy, .field');
  if (host && !host.querySelector('textarea.manual')) {
    const ta = document.createElement('textarea');
    ta.className = 'manual';
    ta.value = text;
    ta.setAttribute('readonly', '');
    host.appendChild(ta);
    ta.focus();
    ta.select();
  }
  done('Copy from the white box');
}

document.addEventListener('click', (e) => {
  const b = e.target.closest('button[data-copy], button[data-news-copy]');
  if (!b) return;
  const art = b.closest('article');
  if (b.dataset.copy) {
    const p = D.posts.find((x) => x.n === art.dataset.n);
    return copyPlain(p[b.dataset.copy], b);
  }
  const x = D.news.find((n) => String(n.n) === art.dataset.news);
  const k = b.dataset.newsCopy;
  const m = k.match(/^subject(\d)$/);
  return copyPlain(m ? x.subjects[Number(m[1])] : x[k], b);
});

document.addEventListener('change', (e) => {
  if (!e.target.matches('label.done input')) return;
  const art = e.target.closest('article');
  write('gdc-desk-posted-' + art.dataset.n, e.target.checked);
  art.classList.toggle('is-done', e.target.checked);
  const live = D.posts.filter((p) => !read(pkey(p)));
  document.getElementById('count').textContent =
    live.length + ' of ' + D.posts.length + ' still to go. Newest at the top, because the oldest goes stale soonest.';
});
